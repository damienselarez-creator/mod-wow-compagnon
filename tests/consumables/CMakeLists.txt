cmake_minimum_required(VERSION 3.16)
project(companion_consumables LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/CompanionConsumables.cpp" source)
string(FIND "${source}" "using ConsumableKind" first)
string(SUBSTRING "${source}" ${first} -1 production)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumables.inc" "${production}")
add_executable(consumables_test consumables_test.cpp)
target_compile_features(consumables_test PRIVATE cxx_std_20)
target_include_directories(consumables_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../../src/gameplay/Bot)
add_test(NAME companion_consumable_selection_and_use COMMAND consumables_test)
