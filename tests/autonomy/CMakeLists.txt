cmake_minimum_required(VERSION 3.16)
project(companion_autonomy LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/CompanionErrands.cpp" source)
function(extract first last output)
    string(FIND "${source}" "${first}" begin)
    string(FIND "${source}" "${last}" end)
    math(EXPR size "${end}-${begin}")
    string(SUBSTRING "${source}" ${begin} ${size} block)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${block}")
endfunction()
extract("    Trainer::Spell const* ChooseLesson(" "bool CanTrainCompanionAt(" "lessons.inc")
extract("bool CanSellCompanionItem(" "bool IsManagedCompanion(" "sales.inc")
extract("    bool HasLearnedLesson(" "    bool CanVisitTrainer(" "verification.inc")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Ai/World/Rpg/Action/RpgSubActions.cpp" source)
extract("bool RpgTrainAction::Execute(" "bool RpgTrainAction::isUseful(" "routing.inc")
foreach(suite training sales routing)
    add_executable(${suite}_test ${suite}_test.cpp)
    target_compile_features(${suite}_test PRIVATE cxx_std_20)
    target_include_directories(${suite}_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME companion_${suite} COMMAND ${suite}_test)
endforeach()
