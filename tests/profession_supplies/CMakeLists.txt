cmake_minimum_required(VERSION 3.16)
project(companion_profession_supplies_tests LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/CompanionErrands.cpp" source)
string(FIND "${source}" "    struct ProfessionSupplies" first)
string(FIND "${source}" "    uint32 RepairEquipment" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} selection)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_supplies.inc" "${selection}")
add_executable(profession_supplies_test profession_supplies_test.cpp)
target_compile_features(profession_supplies_test PRIVATE cxx_std_17)
target_include_directories(profession_supplies_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    ../../src/gameplay/Bot)
add_test(NAME companion_profession_supply_purchases COMMAND profession_supplies_test)
