function(extract first last output)
    string(FIND "${source}" "${first}" begin)
    string(FIND "${source}" "${last}" end)
    math(EXPR size "${end}-${begin}")
    string(SUBSTRING "${source}" ${begin} ${size} block)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${block}")
endfunction()
