cmake_minimum_required(VERSION 3.16)
project(companion_travel_tests LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/CompanionErrands.cpp" source)
string(FIND "${source}" "    bool TeachesRiding" first)
string(FIND "${source}" "    class MountItemUse" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} helpers)
string(FIND "${source}" "    PoisonPurchase ChooseMount" first)
string(FIND "${source}" "    struct ProfessionSupplies" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} selection)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_travel.inc" "${helpers}\n${selection}")
add_executable(travel_test travel_test.cpp)
target_compile_features(travel_test PRIVATE cxx_std_17)
target_include_directories(travel_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../../src/gameplay/Bot)
add_test(NAME companion_riding_mount_purchase_and_gather_leash COMMAND travel_test)
