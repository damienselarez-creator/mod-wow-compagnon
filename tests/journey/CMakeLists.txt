cmake_minimum_required(VERSION 3.16)
project(companion_journey LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Script/CompanionJourney.cpp" source)
string(FIND "${source}" "    bool Ready" first)
string(FIND "${source}" "    void Arm" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} predicates)
string(FIND "${source}" "    void Process" first)
string(FIND "${source}" "\n}\n\nclass CompanionJourneyPlayerScript" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} process)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production.inc" "${predicates}\n${process}")
add_executable(journey_test journey_test.cpp)
target_compile_features(journey_test PRIVATE cxx_std_20)
target_include_directories(journey_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME confirmed_teleport_and_inn_binding COMMAND journey_test)
