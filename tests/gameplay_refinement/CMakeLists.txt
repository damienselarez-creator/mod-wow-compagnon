cmake_minimum_required(VERSION 3.16)
project(companion_gameplay_refinement LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/CompanionCombat.cpp" source)
string(FIND "${source}" "namespace\n{" begin)
string(SUBSTRING "${source}" ${begin} -1 source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/combat.inc" "${source}")
add_executable(gameplay_refinement_test gameplay_refinement_test.cpp)
target_compile_features(gameplay_refinement_test PRIVATE cxx_std_20)
target_include_directories(gameplay_refinement_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../../src/gameplay/Bot)
add_test(NAME companion_combat_coordination_and_activity_arbitration COMMAND gameplay_refinement_test)
