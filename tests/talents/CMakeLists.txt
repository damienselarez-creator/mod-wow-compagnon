cmake_minimum_required(VERSION 3.16)
project(companion_talents LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/Factory/PlayerbotFactory.cpp" source)
function(extract first last output)
    string(FIND "${source}" "${first}" begin)
    string(FIND "${source}" "${last}" end)
    if(begin LESS 0 OR end LESS begin)
        message(FATAL_ERROR "Talent source boundaries missing: ${output}")
    endif()
    math(EXPR size "${end}-${begin}")
    string(SUBSTRING "${source}" ${begin} ${size} block)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${block}")
endfunction()
extract("uint32 PlayerbotFactory::InitTalentsTree(" "void PlayerbotFactory::InitTalentsBySpecNo(" "tree.inc")
extract("void PlayerbotFactory::InitCompanionTalents(" "ObjectGuid PlayerbotFactory::GetRandomBot(" "allocation.inc")
add_executable(talents_test talents_test.cpp)
target_compile_features(talents_test PRIVATE cxx_std_20)
target_include_directories(talents_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME companion_single_tree_allocation COMMAND talents_test)
