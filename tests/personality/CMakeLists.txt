cmake_minimum_required(VERSION 3.16)
project(pbc_personality_tests LANGUAGES CXX)
enable_testing()
add_executable(personality_test personality_test.cpp)
target_compile_features(personality_test PRIVATE cxx_std_20)
target_include_directories(personality_test PRIVATE ../../src/narrative/character ../../src/narrative/llm ../../deps)
add_test(NAME permanent_personality COMMAND personality_test
    "${CMAKE_CURRENT_SOURCE_DIR}/../../knowledge/personality-traits.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/narrative/character/pbc_personality.cpp" production)
string(FIND "${production}" "bool Save(" first)
string(FIND "${production}" "bool Command(" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${production}" ${first} ${size} save_function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_save.inc" "${save_function}")
add_executable(personality_store_test personality_store_test.cpp)
target_compile_features(personality_store_test PRIVATE cxx_std_20)
target_include_directories(personality_store_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    ../../src/narrative/character ../../src/narrative/llm ../../deps)
add_test(NAME durable_personality COMMAND personality_store_test "${CMAKE_CURRENT_BINARY_DIR}/private-store")
