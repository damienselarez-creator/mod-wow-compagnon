cmake_minimum_required(VERSION 3.16)
project(companion_errand_retry LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay/Bot/CompanionErrands.cpp" source)
function(extract first last output)
    string(FIND "${source}" "${first}" begin)
    string(FIND "${source}" "${last}" end)
    math(EXPR size "${end}-${begin}")
    string(SUBSTRING "${source}" ${begin} ${size} block)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${block}")
endfunction()
extract("    void FinishVendorVisit(" "    Trainer::Spell const* ChooseLesson(" "finish.inc")
extract("void RequestCompanionTraining(" "bool IsCompanionErrandActive(" "update.inc")
add_executable(errand_retry_test errand_retry_test.cpp)
target_compile_features(errand_retry_test PRIVATE cxx_std_20)
target_include_directories(errand_retry_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME companion_errand_retry COMMAND errand_retry_test)
