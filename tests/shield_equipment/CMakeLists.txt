cmake_minimum_required(VERSION 3.16)
project(companion_shield_equipment LANGUAGES CXX)
enable_testing()
set(gameplay "${CMAKE_CURRENT_SOURCE_DIR}/../../src/gameplay")
file(READ "${gameplay}/Bot/CompanionEquipment.h" source)
string(FIND "${source}" "inline bool PrefersCompanionShield" first)
string(FIND "${source}" "#endif" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} policy)
file(READ "${gameplay}/Ai/Base/Actions/EquipAction.cpp" source)
string(FIND "${source}" "void EquipAction::EquipItems" first)
string(FIND "${source}" "// Return bagslot" last)
math(EXPR size "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${size} equip)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production.inc" "${policy}\n${equip}")
add_executable(shield_equipment_test shield_equipment_test.cpp)
target_compile_features(shield_equipment_test PRIVATE cxx_std_20)
target_include_directories(shield_equipment_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME protection_layout_and_equipment_order COMMAND shield_equipment_test)
